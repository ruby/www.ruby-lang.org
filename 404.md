  if (!match || match[1] === 'en' || !languages.includes(match[1])) return;
  const notFound = '/' + match[1] + '/404.html';

    .then(function(found) {
      window.location.replace(found ? english + window.location.search + window.location.hash : notFound);
    });
})();
